// ─── ANIMATION LOOP ───────────────────────────────────────────────────────────

const clock = new THREE.Clock();

function animate() {
  requestAnimationFrame(animate);
  const t = clock.getElapsedTime();

  // Camera
  camera.position.x = camDist * Math.sin(camTheta) * Math.sin(camPhi);
  camera.position.y = camDist * Math.cos(camPhi) + 1.5;
  camera.position.z = camDist * Math.cos(camTheta) * Math.sin(camPhi);
  camera.lookAt(0, 1.5, 0);
  if (!isDragging) camTheta += 0.0006;

  // Chamber rotations — each axis and speed distinct per layer
  // D1 rotates slowly — it is stable, strategic
  chamberMeshes.D1.rotation.y = t * 0.04;
  chamberMeshes.D1.rotation.x = t * 0.015;
  chamberWires.D1.rotation.y  = t * 0.04;
  chamberWires.D1.rotation.x  = t * 0.015;

  // H4 medium rotation
  chamberMeshes.H4.rotation.y = t * -0.07;
  chamberMeshes.H4.rotation.z = t * 0.025;
  chamberWires.H4.rotation.y  = t * -0.07;
  chamberWires.H4.rotation.z  = t * 0.025;

  // M15 faster — tactical layer is more reactive
  chamberMeshes.M15.rotation.y = t * 0.12;
  chamberMeshes.M15.rotation.x = t * -0.04;
  chamberWires.M15.rotation.y  = t * 0.12;
  chamberWires.M15.rotation.x  = t * -0.04;

  // M5 fastest — the live doorway, almost trembling with low integrity
  const m5s = SCENE.geometry.m5.render.scale * 0.70;
  const m5pulse = 1.0 + Math.sin(t * SCENE.geometry.m5.render.pulseRate * Math.PI * 2) * 0.07;
  chamberMeshes.M5.rotation.y = t * -0.22;
  chamberMeshes.M5.rotation.z = t * 0.09;
  chamberWires.M5.rotation.y  = t * -0.22;
  chamberWires.M5.rotation.z  = t * 0.09;
  chamberMeshes.M5.scale.setScalar(m5s * m5pulse);
  chamberWires.M5.scale.setScalar(m5s * m5pulse * 1.05);

  // Thread pulse — the signal breathes
  threadGroup.children.forEach((child, i) => {
    if (child.material && i > 0) {
      child.material.opacity = 0.07 + Math.sin(t * 1.4 + i) * 0.04;
    }
  });

  // Evidence nodes orbit at their home chamber altitude
  evidenceObjects.forEach(obj => {
    const d = obj.userData;
    d.orbitAngle += d.orbitSpeed * 0.007;
    obj.position.x = Math.cos(d.orbitAngle) * d.orbitRadius;
    obj.position.z = Math.sin(d.orbitAngle) * d.orbitRadius;
    obj.position.y = d.orbitBaseY + Math.sin(t * 0.4 + d.orbitAngle) * 0.15;
  });

  // Friction spikes — animated ones pulse in intensity
  frictionSpikes.forEach((spike, i) => {
    if (spike.userData.animated) {
      spike.material.opacity = 0.6 + Math.sin(t * 2.5 + i) * 0.35;
    }
  });

  // Gate rings rotate
  gateGroup.children.forEach((child, i) => {
    if (child.userData.isGateRing) {
      child.rotation.z = t * (0.3 + i * 0.15);
      child.rotation.x = Math.PI/2 + Math.sin(t * 0.2 + i) * 0.2;
    }
  });
  gateSphere.material.emissiveIntensity =
    SCENE.operatorGate.render.illuminated ? 1.8 + Math.sin(t * 1.8) * 0.4 : 0.2;

  // Light drift
  cyanLight.position.x = Math.sin(t * 0.35) * 4;
  cyanLight.position.z = Math.cos(t * 0.25) * 3;

  renderer.render(scene, camera);
}

animate();

window.addEventListener('resize', () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});
