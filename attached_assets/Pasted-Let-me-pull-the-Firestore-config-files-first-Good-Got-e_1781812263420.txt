Block 3 — replace the two dispatch routes in server.ts:

// In-memory cache — seeded from Firestore on boot, kept in sync on writes
const activeDispatches: any[] = [];

// Seed cache from Firestore on startup (last 50 dispatches, newest first)
async function seedDispatchCache() {
  try {
    const snapshot = await db
      .collection(DISPATCHES_COLLECTION)
      .orderBy("timestamp", "desc")
      .limit(50)
      .get();
    snapshot.forEach(doc => activeDispatches.push(doc.data()));
    console.log(`CRYSTAL BRIDGE: ${activeDispatches.length} dispatch records loaded from Firestore ledger.`);
  } catch (err) {
    console.warn("CRYSTAL BRIDGE: Firestore seed failed — operating on empty cache.", err);
  }
}

// POST /api/dispatch — persist to Firestore, update cache
app.post("/api/dispatch", async (req, res) => {
  try {
    const { actionId, operatorName, signature, checksum, asset, directive } = req.body;

    if (!signature || !operatorName) {
      return res.status(400).json({ error: "Operator signature and credential verification details required." });
    }

    const dispatchId = `DISPATCH-${Date.now()}-${Math.random().toString(36).substring(2, 6).toUpperCase()}`;

    const dispatchRecord = {
      id: dispatchId,
      timestamp: new Date().toISOString(),
      actionId: actionId || "FIELD-POSTURE-01",
      operatorName,
      signature: signature.toUpperCase(),
      checksum: checksum || null,
      asset: asset || "BTC",
      directive: directive || "Operational Posture Shift Approved",
      status: "COMPLETED_DISPATCHED",
      origin: "CRYSTAL_BRIDGE_GATEWAY",
      engine: "NVIDIA DGX H100 EXECUTION UNIT"
    };

    // Write to Firestore — use the dispatchId as the document ID for idempotent lookup
    await db
      .collection(DISPATCHES_COLLECTION)
      .doc(dispatchId)
      .set({
        ...dispatchRecord,
        timestamp: FieldValue.serverTimestamp(), // use server timestamp in Firestore
      });

    // Keep cache in sync (prepend — newest first)
    activeDispatches.unshift(dispatchRecord);

    return res.json({
      success: true,
      record: dispatchRecord,
      message: "Gateway: Crystal Bridge dispatch committed to sovereign ledger (Firestore) and execution flow initialized.",
      ledger: "FIRESTORE_PERSISTED"
    });
  } catch (error: any) {
    console.error("Dispatch API error:", error);
    return res.status(500).json({ error: error.message || "Internal Bridge Gateway disruption." });
  }
});

// GET /api/dispatches — serve from cache (seeded from Firestore on boot)
app.get("/api/dispatches", (req, res) => {
  return res.json({ dispatches: activeDispatches });
});

Behaviour after this change:

On boot, the server loads the last 50 dispatches from Firestore into the cache — so GET /api/dispatches is instant and doesn’t hit Firestore on every request. Every POST /api/dispatch writes atomically to Firestore first, then updates the cache. If the server crashes and restarts, the ledger is fully intact. The dispatch document ID matches the id field, so you can do idempotent lookups by ID later if needed.
